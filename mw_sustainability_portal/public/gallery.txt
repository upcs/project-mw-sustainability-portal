function nextImage(){
	var currImg = document.getElementById('mainImage');
	for (var i = 0; i < imgArray.length; i++){
		if(i == imgArray.length-1){
				document.getElementById('mainImage').src = imgArray[0].src; break; }
		if(imgArray[i].src == currImg.src){
			document.getElementById('mainImage').src = imgArray[i+1].src; break;}
	}					
			
}

function prevImage(){
	var currImg = document.getElementById('mainImage');
	for (var i = 0; i < imgArray.length; i++){
		if(imgArray[i].src == currImg.src){
			if(i == 0){
				document.getElementById('mainImage').src = imgArray[imgArray.length-1].src; break; }
			document.getElementById('mainImage').src = imgArray[i-1].src; break;}
	}					
			
}
